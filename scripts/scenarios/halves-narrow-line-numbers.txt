# Две половины редактора, любую сузить разделителем до предела: ошибок нет, ни в списке сообщений, ни в логе (#36)
waitfor state 120000 proj/
node double main.asm
waitfor state 3000 *main.asm (writable)
node right a.txt
tap menu:Open in Right
waitfor state 3000 right: *a.txt (writable)
wait 500
# правая половина сжата до предела
dragby tap split:halves 800 0
wait 800
expect state messages: -
expect not log IllegalArgumentException
# левая половина сжата до предела
dragby tap split:halves -2000 0
wait 800
expect state messages: -
expect not log IllegalArgumentException
shot halves-left-min
